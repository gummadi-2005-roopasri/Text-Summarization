  <script>
    // ========== CONFIG & SDK ==========
    const defaultConfig = {
      background_color: '#0f172a',
      surface_color: '#1e293b',
      text_color: '#e2e8f0',
      primary_action_color: '#6366f1',
      secondary_action_color: '#334155',
      font_family: 'DM Sans',
      font_size: 14,
      app_title: 'Review',
      login_heading: 'Welcome Back',
      summary_heading: 'Text Summarizer',
      skip_login_text: 'Continue as Guest'
    };

    function applyConfig(config) {
      const bg = config.background_color || defaultConfig.background_color;
      const surface = config.surface_color || defaultConfig.surface_color;
      const text = config.text_color || defaultConfig.text_color;
      const primary = config.primary_action_color || defaultConfig.primary_action_color;
      const secondary = config.secondary_action_color || defaultConfig.secondary_action_color;
      const font = config.font_family || defaultConfig.font_family;
      const baseSize = config.font_size || defaultConfig.font_size;

      const app = document.getElementById('app');
      if (app) app.style.backgroundColor = bg;

      document.querySelectorAll('[style*="background-color:#1e293b"], [style*="background-color: #1e293b"]').forEach(el => {
        if (!el.style.background?.includes('gradient')) el.style.backgroundColor = surface;
      });

      document.querySelectorAll('[style*="color:#e2e8f0"], [style*="color: #e2e8f0"], [style*="color:#f1f5f9"]').forEach(el => {
        el.style.color = text;
      });

      document.querySelectorAll('[style*="background-color:#334155"]').forEach(el => {
        el.style.backgroundColor = secondary;
      });

      const title = document.getElementById('appTitle');
      if (title) {
        title.textContent = config.app_title || defaultConfig.app_title;
        title.style.color = text;
        title.style.fontFamily = `'Playfair Display', ${font}, serif`;
        title.style.fontSize = `${baseSize * 2.1}px`;
      }

      const loginH = document.getElementById('loginHeading');
      if (loginH) {
        loginH.textContent = config.login_heading || defaultConfig.login_heading;
        loginH.style.color = text;
        loginH.style.fontSize = `${baseSize * 1.4}px`;
      }

      const sumTitle = document.getElementById('summaryTitle');
      if (sumTitle) {
        sumTitle.textContent = config.summary_heading || defaultConfig.summary_heading;
        sumTitle.style.color = text;
        sumTitle.style.fontFamily = `'Playfair Display', ${font}, serif`;
        sumTitle.style.fontSize = `${baseSize * 1.3}px`;
      }

      const skipLoginBtn = document.getElementById('skipLoginBtn');
      if (skipLoginBtn) {
        skipLoginBtn.textContent = config.skip_login_text || defaultConfig.skip_login_text;
        skipLoginBtn.style.color = text;
      }

      document.body.style.fontFamily = `${font}, 'DM Sans', sans-serif`;

      document.querySelectorAll('input, textarea, select').forEach(el => {
        el.style.backgroundColor = bg;
        el.style.color = text;
        el.style.fontSize = `${baseSize}px`;
      });

      const loginBtn = document.getElementById('loginBtn');
      if (loginBtn) loginBtn.style.background = `linear-gradient(135deg, ${primary}, ${adjustColor(primary, 30)})`;

      const sumBtn = document.getElementById('summarizeBtn');
      if (sumBtn) sumBtn.style.background = `linear-gradient(135deg, ${primary}, ${adjustColor(primary, 30)})`;
    }

    function adjustColor(hex, amount) {
      hex = hex.replace('#', '');
      let r = Math.min(255, parseInt(hex.substring(0,2), 16) + amount);
      let g = Math.min(255, parseInt(hex.substring(2,4), 16) + amount);
      let b = Math.min(255, parseInt(hex.substring(4,6), 16) + amount);
      return `#${r.toString(16).padStart(2,'0')}${g.toString(16).padStart(2,'0')}${b.toString(16).padStart(2,'0')}`;
    }

    function createMutable(key) {
      return {
        get: () => (window.elementSdk?.config?.[key]) || defaultConfig[key],
        set: (v) => { window.elementSdk.setConfig({ [key]: v }); }
      };
    }

    if (window.elementSdk) {
      window.elementSdk.init({
        defaultConfig,
        onConfigChange: async (config) => applyConfig(config),
        mapToCapabilities: (config) => ({
          recolorables: [
            createMutable('background_color'),
            createMutable('surface_color'),
            createMutable('text_color'),
            createMutable('primary_action_color'),
            createMutable('secondary_action_color')
          ],
          borderables: [],
          fontEditable: createMutable('font_family'),
          fontSizeable: createMutable('font_size')
        }),
        mapToEditPanelValues: (config) => new Map([
          ['app_title', config.app_title || defaultConfig.app_title],
          ['login_heading', config.login_heading || defaultConfig.login_heading],
          ['summary_heading', config.summary_heading || defaultConfig.summary_heading],
          ['skip_login_text', config.skip_login_text || defaultConfig.skip_login_text]
        ])
      });
    }

      const pw = document.getElementById('password');
      const isHidden = pw.type === 'password';

      lucide.createIcons();
    });

    // Login
    function goToSummarizer(email = null) {
      if (email) {
        document.getElementById('userEmail').textContent = email;
      } else {
        document.getElementById('userEmail').textContent = 'Guest';
      }

      document.getElementById('loginScreen').style.opacity = '0';
      document.getElementById('loginScreen').style.transition = 'opacity 0.3s ease';

      setTimeout(() => {
        document.getElementById('loginScreen').classList.add('hidden');
        document.getElementById('summarizerScreen').classList.remove('hidden');
        document.getElementById('summarizerScreen').classList.add('fade-in');
        lucide.createIcons();
      }, 300);
    }

    document.getElementById('loginBtn').addEventListener('click', function(e) {
      e.preventDefault();
      const email = document.getElementById('email').value.trim();
      const pw = document.getElementById('password').value.trim();

      if (!email || !pw) {
        document.getElementById('loginError').classList.remove('hidden');
        return;
      }

      document.getElementById('loginError').classList.add('hidden');
      goToSummarizer(email);
    });
